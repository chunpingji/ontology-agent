def _canonical_consequent_hash(consequent: dict | None) -> str:
    """稳定哈希：键排序 + 保留非 ASCII，用于判定 consequent 是否与某已知基线逐字节一致。"""
    return hashlib.sha256(
        json.dumps(consequent or {}, sort_keys=True, ensure_ascii=False).encode("utf-8")
    ).hexdigest()
